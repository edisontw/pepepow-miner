cmake_minimum_required(VERSION 3.20)
project(pepepow_job_state_tests LANGUAGES CXX)
enable_testing()
add_executable(test_job_state ../test_job_state.cpp)
target_compile_features(test_job_state PRIVATE cxx_std_20)
target_include_directories(test_job_state PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../include")
if(MSVC)
  target_compile_options(test_job_state PRIVATE /W4 /WX)
else()
  target_compile_options(test_job_state PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME job_state COMMAND test_job_state)
