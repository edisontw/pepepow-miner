cmake_minimum_required(VERSION 3.20)
project(miner_preflight LANGUAGES CXX)
enable_testing()
add_executable(miner_preflight ../../tools/miner_preflight.cpp)
target_include_directories(miner_preflight PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../include")
target_compile_features(miner_preflight PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(miner_preflight PRIVATE /W4 /WX)
else()
  target_compile_options(miner_preflight PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME local_endpoint_ok COMMAND miner_preflight --check-local-endpoint 127.0.0.1:3333)
set_tests_properties(local_endpoint_ok PROPERTIES PASS_REGULAR_EXPRESSION "\"network_opened\":false")
add_test(NAME localhost_alias_ok COMMAND miner_preflight --check-local-endpoint localhost:3333)
set_tests_properties(localhost_alias_ok PROPERTIES PASS_REGULAR_EXPRESSION "\\\"endpoint\\\":\\\"127[.]0[.]0[.]1:3333\\\"")
foreach(endpoint IN ITEMS 0.0.0.0:3333 127.0.0.2:3333 example.com:3333 127.0.0.1:0 127.0.0.1:65536 127.0.0.1:+80 127.0.0.1:80x localhost.evil:3333 localhost:0 localhost:65536 localhost:+80 localhost:80x)
  string(MAKE_C_IDENTIFIER "${endpoint}" name)
  add_test(NAME invalid_${name} COMMAND miner_preflight --check-local-endpoint "${endpoint}")
  set_tests_properties(invalid_${name} PROPERTIES WILL_FAIL TRUE)
endforeach()

# Explicit local subscribe/authorize smoke; never contacts external pools.
find_package(Threads REQUIRED)
add_executable(test_local_handshake test_local_handshake.cpp)
target_compile_features(test_local_handshake PRIVATE cxx_std_20)
target_include_directories(test_local_handshake PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../include")
target_link_libraries(test_local_handshake PRIVATE Threads::Threads)
if(WIN32)
  target_link_libraries(test_local_handshake PRIVATE ws2_32)
  target_link_libraries(miner_preflight PRIVATE ws2_32)
endif()
if(MSVC)
  target_compile_options(test_local_handshake PRIVATE /W4 /WX)
else()
  target_compile_options(test_local_handshake PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME local_handshake_loopback COMMAND test_local_handshake)
set_tests_properties(local_handshake_loopback PROPERTIES TIMEOUT 10)
add_test(NAME local_handshake_reject_remote COMMAND miner_preflight --local-handshake 0.0.0.0:3333 WALLET.worker)
set_tests_properties(local_handshake_reject_remote PROPERTIES WILL_FAIL TRUE)
