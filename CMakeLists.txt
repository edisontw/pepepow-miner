cmake_minimum_required(VERSION 3.20)
project(pepepow_kat C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

option(HARNESS_CONTRACT_TESTS_ONLY "Build synthetic harness contract test (NOT HooHash)" OFF)
set(PEPEPOW_REFERENCE_LIBRARY "" CACHE FILEPATH "Reviewed library exporting hoohashv110")
set(PEPEPOW_CORE_SOURCE_DIR "" CACHE PATH "Pinned external PePe-core checkout root (not vendored)")

if(HARNESS_CONTRACT_TESTS_ONLY)
  if(PEPEPOW_REFERENCE_LIBRARY OR PEPEPOW_CORE_SOURCE_DIR)
    message(FATAL_ERROR "Synthetic contract mode must not mix with a real reference")
  endif()
  add_executable(kat-contract-test tools/kat.c tests/harness/synthetic_reference.c)
else()
  if(PEPEPOW_REFERENCE_LIBRARY AND PEPEPOW_CORE_SOURCE_DIR)
    message(FATAL_ERROR "Choose exactly one reference source: library OR Core checkout")
  elseif(PEPEPOW_CORE_SOURCE_DIR)
    set(CORE_CRYPTO "${PEPEPOW_CORE_SOURCE_DIR}/src/crypto")
    foreach(source IN ITEMS hoohash/hoohash.c blake3.c blake3_dispatch.c blake3_portable.c)
      if(NOT EXISTS "${CORE_CRYPTO}/${source}")
        message(FATAL_ERROR "Missing pinned Core source: ${CORE_CRYPTO}/${source}")
      endif()
    endforeach()
    # Do not silently use platform libm: Core consensus requires reviewed FP64 semantics.
    find_path(OPENLIBM_INCLUDE_DIR NAMES openlibm_math.h REQUIRED)
    find_library(OPENLIBM_LIBRARY NAMES openlibm REQUIRED)
    add_library(pepepow_core_reference STATIC
      "${CORE_CRYPTO}/hoohash/hoohash.c"
      "${CORE_CRYPTO}/blake3.c"
      "${CORE_CRYPTO}/blake3_dispatch.c"
      "${CORE_CRYPTO}/blake3_portable.c")
    target_include_directories(pepepow_core_reference PRIVATE
      "${CORE_CRYPTO}" "${OPENLIBM_INCLUDE_DIR}")
    target_compile_definitions(pepepow_core_reference PRIVATE HAVE_OPENLIBM=1)
    if(MSVC)
      target_compile_options(pepepow_core_reference PRIVATE /fp:strict)
    elseif(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang|AppleClang")
      target_compile_options(pepepow_core_reference PRIVATE -fno-fast-math -ffp-contract=off)
    endif()
    target_link_libraries(pepepow_core_reference PUBLIC "${OPENLIBM_LIBRARY}")
    set(PEPEPOW_REFERENCE_LIBRARY pepepow_core_reference)
  elseif(NOT PEPEPOW_REFERENCE_LIBRARY OR NOT EXISTS "${PEPEPOW_REFERENCE_LIBRARY}")
    message(FATAL_ERROR "Supply a reviewed PEPEPOW_REFERENCE_LIBRARY or PEPEPOW_CORE_SOURCE_DIR; no synthetic fallback")
  endif()
  add_executable(pepepow-kat tools/kat.c)
  target_link_libraries(pepepow-kat PRIVATE "${PEPEPOW_REFERENCE_LIBRARY}")
endif()
