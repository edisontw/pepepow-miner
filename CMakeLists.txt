cmake_minimum_required(VERSION 3.20)
project(pepepow_kat C)
include(CTest)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

option(HARNESS_CONTRACT_TESTS_ONLY "Build synthetic harness contract test (NOT HooHash)" OFF)
option(PEPEPOW_NATIVE_STRATUM_SEARCH "Build the offline Stratum replay with pinned Core search" OFF)
option(PEPEPOW_ENABLE_LIVE_POOL_MINER "Build the explicitly opt-in public Stratum CPU miner" OFF)
set(PEPEPOW_REFERENCE_LIBRARY "" CACHE FILEPATH "Reviewed library exporting hoohashv110")
set(PEPEPOW_CORE_SOURCE_DIR "" CACHE PATH "Pinned external PePe-core checkout root (not vendored)")
set(PEPEPOW_CORE_EXPECTED_SHA "" CACHE STRING "Full 40-hex reviewed Core commit SHA (required with source checkout)")

if(PEPEPOW_NATIVE_STRATUM_SEARCH AND (HARNESS_CONTRACT_TESTS_ONLY OR NOT PEPEPOW_CORE_SOURCE_DIR))
  message(FATAL_ERROR "Native Stratum replay requires the pinned Core source build")
endif()
if(PEPEPOW_ENABLE_LIVE_POOL_MINER AND NOT PEPEPOW_NATIVE_STRATUM_SEARCH)
  message(FATAL_ERROR "The live Pool CPU miner requires the pinned Core native Stratum build")
endif()

if(HARNESS_CONTRACT_TESTS_ONLY)
  if(PEPEPOW_REFERENCE_LIBRARY OR PEPEPOW_CORE_SOURCE_DIR OR PEPEPOW_CORE_EXPECTED_SHA)
    message(FATAL_ERROR "Synthetic contract mode must not mix with a real reference")
  endif()
  add_executable(kat-contract-test tools/kat.c tests/harness/synthetic_reference.c)
else()
  if(PEPEPOW_REFERENCE_LIBRARY AND PEPEPOW_CORE_SOURCE_DIR)
    message(FATAL_ERROR "Choose exactly one reference source: library OR Core checkout")
  elseif(PEPEPOW_CORE_SOURCE_DIR)
    string(LENGTH "${PEPEPOW_CORE_EXPECTED_SHA}" core_sha_length)
    if(NOT PEPEPOW_CORE_EXPECTED_SHA MATCHES "^[0-9a-fA-F]+$" OR NOT core_sha_length EQUAL 40)
      message(FATAL_ERROR "PEPEPOW_CORE_EXPECTED_SHA must be a full 40-character hex commit SHA")
    endif()
    find_package(Git REQUIRED)
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${PEPEPOW_CORE_SOURCE_DIR}" rev-parse --verify HEAD
      RESULT_VARIABLE core_git_status OUTPUT_VARIABLE core_actual_sha ERROR_QUIET
      OUTPUT_STRIP_TRAILING_WHITESPACE)
    if(NOT core_git_status EQUAL 0 OR NOT core_actual_sha STREQUAL PEPEPOW_CORE_EXPECTED_SHA)
      message(FATAL_ERROR "Core checkout HEAD does not match PEPEPOW_CORE_EXPECTED_SHA (actual: ${core_actual_sha})")
    endif()
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${PEPEPOW_CORE_SOURCE_DIR}" status --porcelain --untracked-files=no
      RESULT_VARIABLE core_status_result OUTPUT_VARIABLE core_dirty OUTPUT_STRIP_TRAILING_WHITESPACE)
    if(NOT core_status_result EQUAL 0 OR NOT core_dirty STREQUAL "")
      message(FATAL_ERROR "Core checkout contains tracked modifications; use a clean pinned checkout")
    endif()
    set(CORE_CRYPTO "${PEPEPOW_CORE_SOURCE_DIR}/src/crypto")
    foreach(source IN ITEMS hoohash/hoohash.c blake3.c blake3_dispatch.c blake3_portable.c)
      if(NOT EXISTS "${CORE_CRYPTO}/${source}")
        message(FATAL_ERROR "Missing pinned Core source: ${CORE_CRYPTO}/${source}")
      endif()
    endforeach()
    # Do not silently use platform libm: Core consensus requires reviewed FP64 semantics.
    find_path(OPENLIBM_INCLUDE_DIR NAMES openlibm_math.h REQUIRED)
    find_library(OPENLIBM_LIBRARY NAMES openlibm REQUIRED)
    add_library(pepepow_core_reference STATIC
      "${CORE_CRYPTO}/hoohash/hoohash.c"
      "${CORE_CRYPTO}/blake3.c"
      "${CORE_CRYPTO}/blake3_dispatch.c"
      "${CORE_CRYPTO}/blake3_portable.c")
    target_include_directories(pepepow_core_reference PRIVATE
      "${CORE_CRYPTO}" "${OPENLIBM_INCLUDE_DIR}")
    target_compile_definitions(pepepow_core_reference PRIVATE HAVE_OPENLIBM=1)
    if(MSVC)
      target_compile_options(pepepow_core_reference PRIVATE /fp:strict)
    elseif(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang|AppleClang")
      target_compile_options(pepepow_core_reference PRIVATE -fno-fast-math -ffp-contract=off)
    endif()
    target_link_libraries(pepepow_core_reference PUBLIC "${OPENLIBM_LIBRARY}")
    set(PEPEPOW_REFERENCE_LIBRARY pepepow_core_reference)
  elseif(NOT PEPEPOW_REFERENCE_LIBRARY OR NOT EXISTS "${PEPEPOW_REFERENCE_LIBRARY}")
    message(FATAL_ERROR "Supply a reviewed PEPEPOW_REFERENCE_LIBRARY or PEPEPOW_CORE_SOURCE_DIR; no synthetic fallback")
  endif()
  add_executable(pepepow-kat tools/kat.c)
  target_link_libraries(pepepow-kat PRIVATE "${PEPEPOW_REFERENCE_LIBRARY}")
  enable_language(CXX)
  add_executable(pepepow-offline-search tools/offline_cpu_search.cpp)
  target_compile_features(pepepow-offline-search PRIVATE cxx_std_17)
  target_include_directories(pepepow-offline-search PRIVATE include)
  target_link_libraries(pepepow-offline-search PRIVATE "${PEPEPOW_REFERENCE_LIBRARY}")

  find_package(Threads REQUIRED)
  add_executable(pepepow-cpu-benchmark tools/cpu_benchmark_main.cpp tools/cpu_benchmark.cpp)
  target_compile_features(pepepow-cpu-benchmark PRIVATE cxx_std_20)
  target_include_directories(pepepow-cpu-benchmark PRIVATE include)
  target_link_libraries(pepepow-cpu-benchmark PRIVATE "${PEPEPOW_REFERENCE_LIBRARY}" Threads::Threads)
  if(WIN32)
    target_link_libraries(pepepow-cpu-benchmark PRIVATE psapi)
  endif()

  if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(test_cpu_worker_pool tests/test_cpu_worker_pool.cpp)
    target_compile_features(test_cpu_worker_pool PRIVATE cxx_std_20)
    target_include_directories(test_cpu_worker_pool PRIVATE include)
    target_link_libraries(test_cpu_worker_pool PRIVATE "${PEPEPOW_REFERENCE_LIBRARY}" Threads::Threads)
    add_test(NAME cpu_worker_pool COMMAND test_cpu_worker_pool)
    set_tests_properties(cpu_worker_pool PROPERTIES TIMEOUT 30)
    add_executable(test_cpu_benchmark_accounting tests/test_cpu_benchmark_accounting.cpp)
    target_compile_features(test_cpu_benchmark_accounting PRIVATE cxx_std_20)
    target_include_directories(test_cpu_benchmark_accounting PRIVATE tools)
    add_test(NAME cpu_benchmark_accounting COMMAND test_cpu_benchmark_accounting)
    set_tests_properties(cpu_benchmark_accounting PROPERTIES TIMEOUT 5)
    add_test(NAME cpu_benchmark_cli
      COMMAND "${Python3_EXECUTABLE}"
              "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cpu_benchmark_cli.py"
              "$<TARGET_FILE:pepepow-cpu-benchmark>")
    set_tests_properties(cpu_benchmark_cli PROPERTIES TIMEOUT 15)
  endif()
endif()

if(PEPEPOW_NATIVE_STRATUM_SEARCH)
  find_package(OpenSSL REQUIRED COMPONENTS Crypto)
  find_package(Threads REQUIRED)
  add_executable(offline_stratum_replay tools/offline_stratum_replay.cpp)
  target_compile_features(offline_stratum_replay PRIVATE cxx_std_20)
  target_compile_definitions(offline_stratum_replay PRIVATE PEPEPOW_NATIVE_STRATUM_SEARCH)
  target_include_directories(offline_stratum_replay PRIVATE include)
  target_link_libraries(offline_stratum_replay PRIVATE pepepow_core_reference OpenSSL::Crypto)

  add_executable(offline_stratum_miner tools/offline_stratum_miner.cpp tools/cpu_benchmark.cpp)
  target_compile_features(offline_stratum_miner PRIVATE cxx_std_20)
  target_include_directories(offline_stratum_miner PRIVATE include)
  target_link_libraries(offline_stratum_miner PRIVATE
    pepepow_core_reference OpenSSL::Crypto Threads::Threads)
  if(WIN32)
    target_link_libraries(offline_stratum_miner PRIVATE ws2_32 psapi)
  endif()
  if(MSVC)
    target_compile_options(offline_stratum_miner PRIVATE /W4 /WX)
  else()
    target_compile_options(offline_stratum_miner PRIVATE -Wall -Wextra -Werror -pedantic)
  endif()

  if(PEPEPOW_ENABLE_LIVE_POOL_MINER)
    add_executable(pepepow-live-cpu-miner tools/offline_stratum_miner.cpp tools/cpu_benchmark.cpp)
    target_compile_features(pepepow-live-cpu-miner PRIVATE cxx_std_20)
    target_compile_definitions(pepepow-live-cpu-miner PRIVATE PEPEPOW_ENABLE_LIVE_POOL_MINER=1)
    target_include_directories(pepepow-live-cpu-miner PRIVATE include)
    target_link_libraries(pepepow-live-cpu-miner PRIVATE
      pepepow_core_reference OpenSSL::Crypto Threads::Threads)
    if(WIN32)
      target_link_libraries(pepepow-live-cpu-miner PRIVATE ws2_32 psapi)
    endif()
    if(MSVC)
      target_compile_options(pepepow-live-cpu-miner PRIVATE /W4 /WX)
    else()
      target_compile_options(pepepow-live-cpu-miner PRIVATE -Wall -Wextra -Werror -pedantic)
    endif()
  endif()

  if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(test_native_stratum_search tests/test_native_stratum_search.cpp)
    target_compile_features(test_native_stratum_search PRIVATE cxx_std_20)
    target_include_directories(test_native_stratum_search PRIVATE include)
    target_link_libraries(test_native_stratum_search PRIVATE
      pepepow_core_reference OpenSSL::Crypto Threads::Threads)
    add_test(NAME native_stratum_search COMMAND test_native_stratum_search)
    set_tests_properties(native_stratum_search PROPERTIES TIMEOUT 10)

    add_executable(test_cuda_search_result_decoder tests/test_cuda_search_result_decoder.cpp)
    target_compile_features(test_cuda_search_result_decoder PRIVATE cxx_std_20)
    target_include_directories(test_cuda_search_result_decoder PRIVATE include)
    add_test(NAME cuda_search_result_decoder COMMAND test_cuda_search_result_decoder)
    set_tests_properties(cuda_search_result_decoder PROPERTIES TIMEOUT 5)

    add_test(NAME offline_stratum_miner_loopback
      COMMAND "${Python3_EXECUTABLE}"
              "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_offline_stratum_miner_e2e.py"
              "$<TARGET_FILE:offline_stratum_miner>")
    set_tests_properties(offline_stratum_miner_loopback PROPERTIES TIMEOUT 30)

    add_executable(test_stratum_endpoint tests/test_stratum_endpoint.cpp)
    target_compile_features(test_stratum_endpoint PRIVATE cxx_std_20)
    target_include_directories(test_stratum_endpoint PRIVATE include)
    if(WIN32)
      target_link_libraries(test_stratum_endpoint PRIVATE ws2_32)
    endif()
    if(MSVC)
      target_compile_options(test_stratum_endpoint PRIVATE /W4 /WX)
    else()
      target_compile_options(test_stratum_endpoint PRIVATE -Wall -Wextra -Werror -pedantic)
    endif()
    add_test(NAME stratum_endpoint_contract COMMAND test_stratum_endpoint)

    if(PEPEPOW_ENABLE_LIVE_POOL_MINER)
      add_test(NAME live_pool_miner_loopback
        COMMAND "${Python3_EXECUTABLE}"
                "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_live_pool_miner_loopback.py"
                "$<TARGET_FILE:pepepow-live-cpu-miner>")
      set_tests_properties(live_pool_miner_loopback PROPERTIES TIMEOUT 30)
    endif()
  endif()

endif()
