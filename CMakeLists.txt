cmake_minimum_required(VERSION 3.20)
project(pepepow_kat C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(HARNESS_CONTRACT_TESTS_ONLY "Build the synthetic harness contract test (NOT HooHash)" OFF)
set(PEPEPOW_REFERENCE_LIBRARY "" CACHE FILEPATH "Reviewed library exporting hoohashv110")
if(HARNESS_CONTRACT_TESTS_ONLY)
  add_executable(kat-contract-test tools/kat.c tests/harness/synthetic_reference.c)
else()
  if(NOT PEPEPOW_REFERENCE_LIBRARY OR NOT EXISTS "${PEPEPOW_REFERENCE_LIBRARY}")
    message(FATAL_ERROR "Supply a reviewed PEPEPOW_REFERENCE_LIBRARY; there is no synthetic fallback")
  endif()
  add_executable(pepepow-kat tools/kat.c)
  target_link_libraries(pepepow-kat PRIVATE "${PEPEPOW_REFERENCE_LIBRARY}")
endif()
